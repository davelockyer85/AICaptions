'use client';

import { useState } from 'react';
import { createClient } from '@supabase/supabase-js';

const PRICING_TIERS = [
  {
    id: 'starter',
    name: 'Starter',
    description: 'Ideal for individual streamers and small presentations.',
    priceMonthly: '$15',
    priceAnnual: '$12',
    priceIdMonthly: 'price_starter_monthly_id',
    priceIdAnnual: 'price_starter_annual_id',
    features: [
      '1 Streamer Session at a time',
      'OBS Transparent Overlay',
      'Basic Custom Styling',
      'Standard Latency (~500ms)',
    ],
  },
  {
    id: 'pro',
    name: 'Pro Presenter',
    description: 'For professional event operators and live productions.',
    popular: true,
    priceMonthly: '$39',
    priceAnnual: '$29',
    priceIdMonthly: 'price_pro_monthly_id',
    priceIdAnnual: 'price_pro_annual_id',
    features: [
      'Unlimited Simultaneous Streams',
      'Low Latency Realtime Broadcast (<100ms)',
      'Custom CSS Overlay Engine',
      'Presenter Stage Confidence View',
      'Multi-language Translation Stream',
      'Priority Support',
    ],
  },
];

export default function PricingPage() {
  const [isAnnual, setIsAnnual] = useState(true);
  const [loadingPriceId, setLoadingPriceId] = useState<string | null>(null);

  const handleCheckout = async (priceId: string) => {
    setLoadingPriceId(priceId);
    try {
      const res = await fetch('/api/stripe/checkout', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ priceId }),
      });

      const { url, error } = await res.json();
      if (error) throw new Error(error);

      if (url) window.location.href = url;
    } catch (err: any) {
      alert(err.message || 'Checkout failed');
    } finally {
      setLoadingPriceId(null);
    }
  };

  return (
    <div className="min-h-screen bg-slate-950 text-slate-100 py-16 px-4 sm:px-6 lg:px-8">
      <div className="max-w-5xl mx-auto text-center">
        <h1 className="text-4xl sm:text-5xl font-extrabold tracking-tight bg-gradient-to-r from-blue-400 to-indigo-500 bg-clip-text text-transparent">
          Simple, Transparent Pricing
        </h1>
        <p className="mt-4 text-lg text-slate-400">
          Scale your live captions from individual streams to large enterprise broadcasts.
        </p>

        {/* Billing Toggle */}
        <div className="mt-8 flex justify-center items-center gap-4">
          <span className={`text-sm ${!isAnnual ? 'text-white font-semibold' : 'text-slate-400'}`}>Monthly</span>
          <button
            onClick={() => setIsAnnual(!isAnnual)}
            className="w-14 h-8 bg-indigo-600 rounded-full p-1 transition-colors duration-200 relative"
          >
            <div
              className={`w-6 h-6 bg-white rounded-full transition-transform duration-200 ${
                isAnnual ? 'translate-x-6' : 'translate-x-0'
              }`}
            />
          </button>
          <span className={`text-sm ${isAnnual ? 'text-white font-semibold' : 'text-slate-400'}`}>
            Annual <span className="text-emerald-400 text-xs font-bold font-mono">(SAVE 20%)</span>
          </span>
        </div>
      </div>

      {/* Pricing Cards */}
      <div className="mt-12 max-w-4xl mx-auto grid grid-cols-1 md:grid-cols-2 gap-8">
        {PRICING_TIERS.map((tier) => {
          const priceId = isAnnual ? tier.priceIdAnnual : tier.priceIdMonthly;
          const displayPrice = isAnnual ? tier.priceAnnual : tier.priceMonthly;

          return (
            <div
              key={tier.id}
              className={`relative rounded-2xl bg-slate-900/80 border p-8 flex flex-col justify-between backdrop-blur-sm ${
                tier.popular ? 'border-indigo-500 shadow-xl shadow-indigo-500/10' : 'border-slate-800'
              }`}
            >
              {tier.popular && (
                <span className="absolute -top-3 right-8 bg-indigo-600 text-white text-xs font-bold uppercase tracking-wider px-3 py-1 rounded-full">
                  Most Popular
                </span>
              )}
              <div>
                <h3 className="text-2xl font-bold text-white">{tier.name}</h3>
                <p className="mt-2 text-sm text-slate-400">{tier.description}</p>
                <div className="mt-6 flex items-baseline gap-1">
                  <span className="text-4xl font-extrabold text-white">{displayPrice}</span>
                  <span className="text-slate-400 text-sm">/ month</span>
                </div>

                <ul className="mt-8 space-y-3 text-sm text-slate-300">
                  {tier.features.map((feature, i) => (
                    <li key={i} className="flex items-center gap-3">
                      <svg className="w-5 h-5 text-emerald-400 shrink-0" fill="none" viewBox="0 0 24 24" stroke="currentColor">
                        <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M5 13l4 4L19 7" />
                      </svg>
                      {feature}
                    </li>
                  ))}
                </ul>
              </div>

              <button
                onClick={() => handleCheckout(priceId)}
                disabled={loadingPriceId === priceId}
                className={`mt-8 w-full py-3 px-4 rounded-xl font-semibold transition-all duration-150 ${
                  tier.popular
                    ? 'bg-indigo-600 hover:bg-indigo-500 text-white shadow-lg shadow-indigo-600/25'
                    : 'bg-slate-800 hover:bg-slate-700 text-white'
                } disabled:opacity-50`}
              >
                {loadingPriceId === priceId ? 'Redirecting...' : 'Get Started'}
              </button>
            </div>
          );
        })}
      </div>
    </div>
  );
}
